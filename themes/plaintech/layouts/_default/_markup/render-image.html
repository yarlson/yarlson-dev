{{/* Normalize relative Markdown image paths to the current content section. */}} {{- $src := .Destination -}} {{- $isRemote := or (strings.HasPrefix $src "http://") (strings.HasPrefix $src "https://") -}} {{- if and (not $isRemote) (not (strings.HasPrefix $src "/")) -}} {{- $src = printf "/%s/%s" .Page.Section (strings.TrimPrefix "./" $src) -}} {{- end -}}
