{{/* OpenGraph + Twitter using dynamic PNG generation via Hugo image filters - Requires assets/og_base.png and a font in assets (e.g., /Inter-SemiBold.ttf) - For home: uses site description; for pages: uses page title - Optional .Params.thumbnail (page resource) is overlaid top-right */}} {{- $desc := "" -}} {{- with .Params.description -}} {{- $desc = (. | plainify | truncate 200) -}} {{- else -}} {{- with .Params.summary -}} {{- $desc = (. | plainify | truncate 200) -}} {{- else -}} {{- if .IsPage -}} {{- $desc = (.Summary | plainify | truncate 200) -}} {{- else -}} {{- $desc = (site.Params.description | default site.Title | plainify | truncate 200) -}} {{- end -}} {{- end -}} {{- end -}} {{- $ogTitle := cond .IsHome site.Title .Title -}} {{- $ogType := cond .IsPage "article" "website" -}} {{ with site.Language.LanguageCode }}{{ end }} {{ if .IsPage }} {{ with .Date }}{{ end }} {{ with .Lastmod }}{{ end }} {{ range .Params.tags }}{{ end }} {{ end }} {{ with site.Params.twitter }}{{ end }} {{ with site.Params.twitter }}{{ end }} {{/* If base image missing or not an image, bail safely */}} {{ $base := resources.Get "og_base.png" }} {{ if and $base (eq $base.MediaType.Type "image/png") }} {{ $img := $base }} {{ $img = $img.Process "fill 1200x630" }} {{ $text := $.Title }} {{ if $.IsHome }} {{ $text = (site.Params.description | default site.Title) }} {{ end }} {{ with $.Params.thumbnail }} {{ $thumb := $.Resources.Get . }} {{ with $thumb }} {{ $img = $img.Filter (images.Overlay (.Process "fit 300x250") 875 38 )}} {{ end }} {{ end }} {{ $img = $img.Filter (images.Text $text (dict "color" "#000000" "size" 64 "linespacing" 18 "x" 80 "y" 320 ))}} {{ $img = $img.Filter (images.Text site.Title (dict "color" "#000000" "size" 48 "linespacing" 2 "x" 80 "y" 120 ))}} {{ $img = resources.Copy (path.Join $.RelPermalink "og.png") $img }} {{ else }} {{ end }}
